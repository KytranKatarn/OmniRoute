**Blast radius:** Low. All writes are to new files, new cron entries, and new `backup_log` rows. No existing route or agent behavior changes. Rollback = `rm /etc/cron.d/archie-backup`.

```bash
#!/usr/bin/env bash
# scripts/install_backup_cron.sh
# Idempotent installer for /etc/cron.d/archie-backup.
# Must be run as root (sudo). Safe to re-run.
#
# Converts scripts/archie_crontab.txt (user-crontab format) to
# /etc/cron.d/ format (adds user column) and drops it in place with
# correct perms (root:root 0644).

set -euo pipefail

REPO="/mnt/archie_brain"
SRC="$REPO/scripts/archie_crontab.txt"
DST="/etc/cron.d/archie-backup"
USER_FOR_CRON="kytran"

[[ $EUID -eq 0 ]] || { echo "Must run as root (use sudo)."; exit 2; }
[[ -f "$SRC" ]] || { echo "Missing $SRC"; exit 1; }

# Build /etc/cron.d/ file — skip comments and blanks from source, add user column to each entry.
{
    echo "# /etc/cron.d/archie-backup"
    echo "# Generated by scripts/install_backup_cron.sh on $(date -u +%FT%TZ)"
    echo "# Source: $SRC — edit that file, re-run installer."
    echo "SHELL=/bin/bash"
    echo "PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
    echo "MAILTO=\"\""
    echo ""
    # Each non-comment, non-blank line: insert USER_FOR_CRON after minute/hour/dom/mon/dow.
    awk -v user="$USER_FOR_CRON" '
        /^#/ || NF == 0 { print; next }
        { $6 = user " " $6; print }
    ' "$SRC"
} > "$DST.tmp"

chown root:root "$DST.tmp"
chmod 0644 "$DST.tmp"
mv "$DST.tmp" "$DST"

# Tell cron to reload (most distros watch /etc/cron.d, but a restart is safer).
systemctl reload cron 2>/dev/null || systemctl restart cron

echo "Installed $DST"
ls -la "$DST"
```

Checks: OAuth configured, service builds, folder id resolves, test upload + delete roundtrip."""

def run_in_platform(py: str) -> tuple[int, str]:
    r = subprocess.run(
        ["docker", "exec", "archie_platform", "python3", "-c", py],
        capture_output=True, text=True,

    return r.returncode, (r.stdout + r.stderr).strip()

        "from tools.google_work_suite.services.drive_service import backup_is_oauth_configured;"
        "print(backup_is_oauth_configured())"

        "from tools.google_work_suite.services.drive_service import "
        "backup_get_drive_service, backup_ensure_folder;"
        "svc = backup_get_drive_service();"

    with tempfile.NamedTemporaryFile(prefix="dr_smoke_", suffix=".txt", delete=False) as tf:
        tf.write(f"smoke test {time.time()}\n".encode())
        local = tf.name
    remote = f"dr_smoke_{int(time.time())}.txt"

    rc, out = run_in_platform(
        "from tools.google_work_suite.services.drive_service import ("
        " backup_get_drive_service, backup_ensure_folder,"
        " backup_upload_file, backup_delete_file);"

```python
def _check_backup_staleness(self) -> tuple[str, str]:
    """Alert if most recent successful primary backup is older than 36h.
    Returns (severity, message) or ('ok', '') if healthy."""
    with get_db_cursor() as (_, cur):
        cur.execute("""
            SELECT EXTRACT(EPOCH FROM (NOW() - MAX(created_at))) / 3600.0 AS hours_stale
            FROM backup_log
            WHERE tier='primary' AND status='success' AND component='run_backup'
        """)
        row = cur.fetchone()
    hours = (row and row[0]) or 9999
    if hours > 36:
        return ("error", f"Primary backup stale: {hours:.1f} h since last success")
    if hours > 26:
        return ("warning", f"Primary backup approaching stale: {hours:.1f} h")
    return ("ok", "")

def _check_backup_tier_health(self) -> tuple[str, str]:
    """Warn if any offsite tier (gdrive, vps, usb) has been skipping for >14 days."""
    with get_db_cursor() as (_, cur):
        cur.execute("""
            SELECT tier,
                   EXTRACT(EPOCH FROM (NOW() - MAX(CASE WHEN status='success'
                                                        THEN created_at END))) / 86400.0
                   AS days_since_success
            FROM backup_log
            WHERE tier IN ('usb', 'gdrive', 'vps')
            GROUP BY tier
        """)
        bad = [(t, d) for t, d in cur.fetchall() if d is None or d > 14]
    if not bad:
        return ("ok", "")
    details = ", ".join(f"{t}={'never' if d is None else f'{d:.1f}d'}" for t, d in bad)
    return ("warning", f"Backup tiers stale (>14d): {details}")
```

Register both checks in the watchdog's check list so they run every cycle. Follow the existing dedup pattern so alerts don't spam.

# A.R.C.H.I.E. Backup & Restore Runbook

## TL;DR — Full Rebuild After Catastrophic Loss

1. Fresh Ubuntu host. Install docker, postgresql-client, openssl, git, rsync.
2. Clone repo: `git clone git@github.com:KytranKatarn/archie-platform.git /mnt/archie_brain`
3. Retrieve 3 passphrases from paper safe: BACKUP_CONFIG_PASSPHRASE, BACKUP_GDRIVE_PASSPHRASE, BACKUP_VPS_PASSPHRASE.

6. Build + boot: `docker compose up -d archie_database`
7. Restore DB: `pg_restore -h localhost -U archie -d archie -Fc <latest_db.dump>`
8. Boot platform: `docker compose up -d`
9. Verify: `curl -s http://localhost:3000/health | jq`

## Recovery Time Objective

| Scenario                         | RTO     | Notes |
|----------------------------------|---------|-------|
| Single container crash           | 5 min   | `docker restart` |
| Host disk failure, USB intact    | 3 h     | From USB DB + config + code |
| Host + USB both lost             | 6–8 h   | From GDrive cold (download + decrypt) |
| Hub fire/theft                   | 12–24 h | From VPS Outpost rsync + GDrive redundancy |

All three passphrases MUST be:
- Printed on paper
- Stored in a physical safe
- Off-site (not in the same building as the hub)
- Rotated ONLY when no legacy encrypted backups remain valid

Losing a passphrase = losing every encrypted backup in that tier.

- [ ] **Step 5.2 — Write `scripts/backup/dr_drill.sh`**

```bash
#!/usr/bin/env bash
# Quarterly disaster-recovery drill.
# Restores the latest daily DB dump into a throwaway Postgres container
# and confirms row counts match production within 1%.
# Writes a report to scripts/backup/dr_reports/.
set -euo pipefail

REPO="/mnt/archie_brain"
SRC_DIR="${BACKUP_ROOT:-/mnt/data2/backups}"
REPORT_DIR="$REPO/scripts/backup/dr_reports"
mkdir -p "$REPORT_DIR"
TS=$(date -u +%Y-%m-%d)
REPORT="$REPORT_DIR/$TS.md"
CONTAINER="archie_dr_drill_$$"

latest_dump=$(ls -td "$SRC_DIR"/*/database/*.dump 2>/dev/null | head -1 || true)
[[ -n "${latest_dump:-}" ]] || { echo "No dump found in $SRC_DIR/*/database/"; exit 1; }

echo "DR Drill $TS — using $latest_dump"

# Row counts from production.
prod_rows=$(docker exec archie_database psql -U archie -d archie -At -c \
    "SELECT COUNT(*) FROM knowledge_base;")

# Spin up throwaway container on a random high port.
port=$(( RANDOM % 20000 + 40000 ))
docker run -d --rm --name "$CONTAINER" -e POSTGRES_PASSWORD=drilldrill \
    -p "$port:5432" postgres:15 > /dev/null
sleep 5

# Create DB and restore.
PGPASSWORD=drilldrill psql -h 127.0.0.1 -p "$port" -U postgres -c "CREATE DATABASE archie;" >/dev/null
PGPASSWORD=drilldrill pg_restore -h 127.0.0.1 -p "$port" -U postgres -d archie --no-owner --no-privileges \
    "$latest_dump" 2>&1 | tail -5

drill_rows=$(PGPASSWORD=drilldrill psql -h 127.0.0.1 -p "$port" -U postgres -d archie -At -c \
    "SELECT COUNT(*) FROM knowledge_base;")

docker rm -f "$CONTAINER" >/dev/null

pct_diff=$(awk "BEGIN{printf \"%.2f\", ($prod_rows - $drill_rows) / $prod_rows * 100}")
status="PASS"
awk "BEGIN{exit !(($prod_rows - $drill_rows) / $prod_rows < 0.01 ? 1 : 0)}" || status="FAIL"

cat >"$REPORT" <<EOF
# DR Drill — $TS

- Dump: \`$latest_dump\`
- Production rows (knowledge_base): $prod_rows
- Drill restore rows: $drill_rows
- Delta: $pct_diff%
- Status: **$status**

Acceptance: delta < 1%.
EOF

echo "Report: $REPORT"
echo "Status: $status"
[[ "$status" == "PASS" ]] || exit 1
```
