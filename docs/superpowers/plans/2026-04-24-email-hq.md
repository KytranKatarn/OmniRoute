  ```python
  """Stalwart Mail Admin API wrapper.

  Provisions and manages mailboxes on the Stalwart mail server.
  Internal URL: http://stalwart_mail:8080 (Docker network)
  External URL: http://192.168.1.200:8101
  """
  from __future__ import annotations

  import logging
  import os

  import requests

  logger = logging.getLogger(__name__)

  _BASE = os.getenv("STALWART_ADMIN_URL", "http://stalwart_mail:8080")
  _USER = os.getenv("STALWART_ADMIN_USER", "admin")
  _PASS = os.getenv("STALWART_ADMIN_PASSWORD", "")


  def _auth() -> tuple:
      return (_USER, _PASS)


  def test_connection() -> dict:
      """Check if Stalwart admin API is reachable. Returns {ok, version, error}."""
      try:
          resp = requests.get(f"{_BASE}/api/health", auth=_auth(), timeout=5)
          if resp.status_code == 200:
              return {"ok": True, "status": resp.status_code}
          return {"ok": False, "status": resp.status_code, "error": resp.text[:200]}
      except Exception as exc:
          return {"ok": False, "error": str(exc)}


  def list_accounts() -> list:
      """List all mailbox accounts in Stalwart."""
      try:
          resp = requests.get(
              f"{_BASE}/api/principal",
              auth=_auth(),
              params={"type": "individual"},
              timeout=10,
          )
          if resp.status_code == 200:
              return resp.json().get("items", [])
          logger.error("Stalwart list_accounts failed: %s", resp.status_code)
          return []
      except Exception as exc:
          logger.error("Stalwart list_accounts error: %s", exc)
          return []


  def create_mailbox(address: str, display_name: str = "", password: str = None) -> dict:
      """Provision a new mailbox in Stalwart.

      Returns {ok, error}.
      """
      import secrets
      local_part = address.split("@")[0]
      _password = password or secrets.token_urlsafe(20)

      try:
          resp = requests.post(
              f"{_BASE}/api/principal",
              auth=_auth(),
              json={
                  "type": "individual",
                  "name": local_part,
                  "description": display_name or local_part,
                  "secrets": [_password],
                  "emails": [address],
                  "quota": 1073741824,  # 1 GB default
              },
              timeout=10,
          )
          if resp.status_code in (200, 201):
              logger.info("Stalwart: provisioned mailbox %s", address)
              return {"ok": True}
          err = resp.text[:300]
          logger.error("Stalwart create_mailbox failed for %s: %s", address, err)
          return {"ok": False, "error": err}
      except Exception as exc:
          logger.error("Stalwart create_mailbox error for %s: %s", address, exc)
          return {"ok": False, "error": str(exc)}


  def delete_mailbox(address: str) -> dict:
      """Remove a mailbox from Stalwart. Returns {ok, error}."""
      local_part = address.split("@")[0]
      try:
          resp = requests.delete(
              f"{_BASE}/api/principal/{local_part}",
              auth=_auth(),
              timeout=10,
          )
          if resp.status_code in (200, 204):
              logger.info("Stalwart: deleted mailbox %s", address)
              return {"ok": True}
          return {"ok": False, "error": resp.text[:200]}
      except Exception as exc:
          return {"ok": False, "error": str(exc)}
  ```

  ```python
  """Email HQ — Overview tab routes (stats + activity feed)."""
  import logging
  from database import get_db_cursor
  from flask_login import login_required
  from flask import jsonify

  logger = logging.getLogger(__name__)


  def register_overview_routes(bp, admin_required):

      @bp.route("/api/email/overview/stats", methods=["GET"])
      @login_required
      @admin_required
      def email_overview_stats():
          """Stats: total unread, sent today, agents wired, not provisioned."""
          with get_db_cursor() as (conn, cur):
              cur.execute(
                  """
                  SELECT
                    (SELECT COUNT(*) FROM email_activity_log
                     WHERE direction='received' AND read_at IS NULL) AS unread,
                    (SELECT COUNT(*) FROM email_activity_log
                     WHERE direction='sent'
                       AND logged_at >= NOW() - INTERVAL '24 hours') AS sent_today,
                    (SELECT COUNT(*) FROM email_mailboxes WHERE provisioned = TRUE) AS provisioned,
                    (SELECT COUNT(*) FROM email_mailboxes WHERE provisioned = FALSE AND active = TRUE) AS not_provisioned
                  """
              )
              row = cur.fetchone()
          return jsonify({
              "unread": row["unread"],
              "sent_today": row["sent_today"],
              "provisioned": row["provisioned"],
              "not_provisioned": row["not_provisioned"],
          })

      @bp.route("/api/email/overview/feed", methods=["GET"])
      @login_required
      @admin_required
      def email_overview_feed():
          """Last 50 email activity events across all mailboxes."""
          with get_db_cursor() as (conn, cur):
              cur.execute(
                  """
                  SELECT al.id, al.mailbox_address, al.direction, al.from_address,
                         al.to_address, al.subject, al.logged_at,
                         m.display_name AS mailbox_display
                  FROM email_activity_log al
                  LEFT JOIN email_mailboxes m ON m.address = al.mailbox_address
                  ORDER BY al.logged_at DESC
                  LIMIT 50
                  """,
              )
              rows = [dict(r) for r in cur.fetchall()]
              for r in rows:
                  if r.get("logged_at"):
                      r["logged_at"] = r["logged_at"].isoformat()
          return jsonify({"events": rows})
  ```

  ```python
  """Email HQ — Agents tab: mailbox listing and Stalwart provisioning."""
  import logging
  from database import get_db_cursor
  from flask import jsonify, request
  from flask_login import login_required

  logger = logging.getLogger(__name__)


  def register_mailbox_routes(bp, admin_required):

      @bp.route("/api/email/mailboxes", methods=["GET"])
      @login_required
      @admin_required
      def email_list_mailboxes():
          """List all mailboxes with provisioning status."""
          mailbox_type = request.args.get("type")
          with get_db_cursor() as (conn, cur):
              query = """
                  SELECT m.id, m.address, m.mailbox_type, m.display_name,
                         m.provisioned, m.provisioned_at, m.active,
                         a.name AS agent_name, a.department AS agent_dept
                  FROM email_mailboxes m
                  LEFT JOIN agents a ON a.id = m.owner_agent_id
                  WHERE m.active = TRUE
              """
              params = []
              if mailbox_type:
                  query += " AND m.mailbox_type = %s"
                  params.append(mailbox_type)
              query += " ORDER BY m.mailbox_type, m.address"
              cur.execute(query, params)
              rows = [dict(r) for r in cur.fetchall()]
              for r in rows:
  ```

  ```python
  import logging

  from email.header import decode_header

  from flask_login import login_required

  logger = logging.getLogger(__name__)

  _IMAP_HOST = os.getenv("IMAP_HOST", "192.168.1.200")
  _IMAP_PORT = int(os.getenv("IMAP_PORT", "993"))
  _IMAP_USE_SSL = os.getenv("IMAP_SSL", "true").lower() == "true"


  def _decode_header_value(value: str) -> str:
      """Decode MIME-encoded email header."""
      parts = decode_header(value or "")
      result = []
      for content, charset in parts:
          if isinstance(content, bytes):
              result.append(content.decode(charset or "utf-8", errors="replace"))
          else:
              result.append(content)
      return " ".join(result)


  def _connect_imap(address: str, password: str) -> imaplib.IMAP4_SSL | None:
      """Open an IMAP connection for a given mailbox."""
      try:
          if _IMAP_USE_SSL:
              conn = imaplib.IMAP4_SSL(_IMAP_HOST, _IMAP_PORT)
          else:
              conn = imaplib.IMAP4(_IMAP_HOST, _IMAP_PORT)
          conn.login(address, password)
          return conn
      except Exception as exc:
          logger.error("IMAP connect failed for %s: %s", address, exc)
          return None
  ```
