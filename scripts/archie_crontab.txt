# ==============================================================================
# A.R.C.H.I.E. Crontab Configuration
# ==============================================================================
# Install with: sudo /mnt/archie_brain/scripts/install_backup_cron.sh
# (Installer converts this file to /etc/cron.d/archie-backup and reloads cron.)
#
# Backups route through the new Phase 1-5 pipeline at scripts/backup/:
#   run_backup.sh daily|weekly|monthly  →  per-component scripts
#                                        →  tier_sync_usb + tier_sync_gdrive
#                                        →  retention_prune (covers pre-change cleanup)
#
# Log locations:
#   - Backups: /var/log/archie_backup.log
#   - Memory:  /var/log/archie_memory.log
# Spec: docs/superpowers/specs/2026-04-16-backup-system-design.md
# ==============================================================================

# Daily backup at 02:00 — DB, code, gitbundle, config, volumes + USB/GDrive tier sync
0 2 * * * /mnt/archie_brain/scripts/backup/run_backup.sh daily >> /var/log/archie_backup.log 2>&1

# Weekly backup (Sunday at 03:00) — adds convlogs, kb_export, vps_pull
0 3 * * 0 /mnt/archie_brain/scripts/backup/run_backup.sh weekly >> /var/log/archie_backup.log 2>&1

# Weekly integrity verification (Sunday at 05:00) — after retention prune settles
0 5 * * 0 /mnt/archie_brain/scripts/backup/verify_backups.sh >> /var/log/archie_backup.log 2>&1

# Monthly full backup (1st of month at 01:00) — includes monthly_full tarball
0 1 1 * * /mnt/archie_brain/scripts/backup/run_backup.sh monthly >> /var/log/archie_backup.log 2>&1

# Weekly memory decay (Sunday at 04:00) — agent emotional state decay, not backup-related
0 4 * * 0 /mnt/archie_brain/scripts/weekly_memory_decay.sh >> /var/log/archie_memory.log 2>&1
